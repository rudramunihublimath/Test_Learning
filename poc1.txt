import React, { useState } from 'react';
import { 
  Activity, 
  UploadCloud, 
  PlusCircle, 
  Download, 
  Server, 
  CheckCircle2, 
  XCircle, 
  Settings, 
  FileCode, 
  Send, 
  Info, 
  Layers, 
  ShieldCheck, 
  RefreshCw,
  Terminal,
  HelpCircle,
  Key
} from 'lucide-react';

export default function App() {
  // Config state
  const [baseUrl, setBaseUrl] = useState('http://localhost:8080/api');
  const [useMock, setUseMock] = useState(true);
  const [activeTab, setActiveTab] = useState('updateSpec'); // createApi, updateSpec, downloadSpec, healthCheck

  // Global Notification state
  const [notification, setNotification] = useState(null); // { type: 'success' | 'error', message: '', details: '' }
  const [responseLog, setResponseLog] = useState(null);
  const [loading, setLoading] = useState(false);

  // Form States
  // 1. Create New API
  const [createForm, setCreateForm] = useState({
    apiName: '',
    description: '',
    version: '1.0.0',
    endpointPath: '/api/v1/resource',
    authType: 'Bearer Token'
  });

  // 2. Update Spec API
  const [updateSpecForm, setUpdateSpecForm] = useState({
    xHeaders: 'Yes',
    commitMessage: '',
    spnValue: 'SPN',
    yamlFile: null
  });

  // 3. Download Spec API
  const [downloadForm, setDownloadForm] = useState({
    apiId: '',
    format: 'yaml',
    includeDocs: true
  });

  // API Metadata Dictionary for Center Column Info
  const apiInfoData = {
    createApi: {
      title: 'Create New API',
      method: 'POST',
      endpoint: '/v1/apis/create',
      description: 'Registers a new RESTful API specification into the gateway registry. Initialises route configurations, rate limiting rules, and metadata records.',
      headersRequired: ['Content-Type: application/json', 'Authorization: Bearer <token>'],
      features: [
        'Automatic routing table generation',
        'Payload validation schema creation',
        'Initial API key provisioning'
      ]
    },
    updateSpec: {
      title: 'Update Spec API',
      method: 'POST',
      endpoint: '/v1/apis/spec/update',
      description: 'Uploads and validates an updated OpenAPI/Swagger specification YAML definition. Applies modifications, custom headers, and source control commit messages to Spring Boot registry.',
      headersRequired: ['Content-Type: multipart/form-data', 'X-Custom-Headers: boolean'],
      features: [
        'Multipart binary upload (.yaml / .yml)',
        'Custom Header toggling (xHeaders)',
        'SPN / UAMI security context mapping',
        'Audit trail logging with Commit Message'
      ]
    },
    downloadSpec: {
      title: 'Download Spec API',
      method: 'POST',
      endpoint: '/v1/apis/spec/download',
      description: 'Generates and downloads the published OpenAPI spec document for a given API identifier in YAML or JSON format.',
      headersRequired: ['Content-Type: application/json', 'Accept: application/x-yaml'],
      features: [
        'Export as YAML or JSON',
        'Include optional Swagger UI inline documentation',
        'Synchronised with live gateway configuration'
      ]
    },
    healthCheck: {
      title: 'Health Check',
      method: 'GET',
      endpoint: '/actuator/health',
      description: 'Executes a system status check against the target Spring Boot microservice to verify database connections, gateway responsiveness, and operational health.',
      headersRequired: ['Accept: application/json'],
      features: [
        'Real-time response latency check',
        'Spring Boot Actuator metric probe',
        'Non-intrusive status polling'
      ]
    }
  };

  const showNotification = (type, message, details = null) => {
    setNotification({ type, message, details });
  };

  const clearNotification = () => {
    setNotification(null);
  };

  // 1. Handle Update Spec API Submission (Multipart)
  const handleUpdateSpecSubmit = async (e) => {
    e.preventDefault();
    setLoading(true);
    setResponseLog(null);
    clearNotification();

    if (!updateSpecForm.yamlFile && !useMock) {
      showNotification('error', 'Validation Failed', 'Please select a .yaml / .yml spec file to upload.');
      setLoading(false);
      return;
    }

    try {
      if (useMock) {
        // Mock Server Response Simulation
        await new Promise((resolve) => setTimeout(resolve, 800));
        const mockSuccess = true; // Toggle for testing mock failure if desired

        if (mockSuccess) {
          const resData = {
            status: 200,
            message: 'Specification updated successfully in Spring Boot backend!',
            details: {
              xHeaders: updateSpecForm.xHeaders,
              commitMessage: updateSpecForm.commitMessage || 'Updated OpenAPI schema',
              spnValue: updateSpecForm.spnValue,
              filename: updateSpecForm.yamlFile ? updateSpecForm.yamlFile.name : 'api_spec_v2.yaml',
              timestamp: new Date().toISOString()
            }
          };
          setResponseLog(resData);
          showNotification('success', 'SUCCESS (200 OK)', 'API Specification file updated successfully.');
        } else {
          throw new Error('500 Internal Server Error: Failed to parse YAML file specification.');
        }
      } else {
        // Real Spring Boot fetch call using FormData
        const formData = new FormData();
        formData.append('xHeaders', updateSpecForm.xHeaders);
        formData.append('commitMessage', updateSpecForm.commitMessage);
        formData.append('spnValue', updateSpecForm.spnValue);
        if (updateSpecForm.yamlFile) {
          formData.append('yamlFile', updateSpecForm.yamlFile);
        }

        const response = await fetch(`${baseUrl}/v1/apis/spec/update`, {
          method: 'POST',
          body: formData,
        });

        const contentType = response.headers.get('content-type');
        let data;
        if (contentType && contentType.includes('application/json')) {
          data = await response.json();
        } else {
          data = await response.text();
        }

        setResponseLog(data);

        if (response.ok) {
          showNotification('success', `SUCCESS (${response.status} OK)`, 'Spring Boot process completed successfully.');
        } else {
          showNotification('error', `ERROR (${response.status})`, typeof data === 'string' ? data : data.message || 'Operation failed on Spring Boot backend.');
        }
      }
    } catch (err) {
      setResponseLog({ error: err.message });
      showNotification('error', 'ERROR (Connection Refused)', err.message || 'Could not connect to Spring Boot server.');
    } finally {
      setLoading(false);
    }
  };

  // 2. Handle Create API Submission (JSON POST)
  const handleCreateApiSubmit = async (e) => {
    e.preventDefault();
    setLoading(true);
    setResponseLog(null);
    clearNotification();

    try {
      if (useMock) {
        await new Promise((resolve) => setTimeout(resolve, 800));
        const resData = {
          status: 200,
          message: 'API Created successfully',
          apiId: 'api-' + Math.floor(100000 + Math.random() * 900000),
          payload: createForm,
          createdAt: new Date().toISOString()
        };
        setResponseLog(resData);
        showNotification('success', 'SUCCESS (200 OK)', `API '${createForm.apiName || 'New API'}' created.`);
      } else {
        const response = await fetch(`${baseUrl}/v1/apis/create`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(createForm)
        });
        const data = await response.json();
        setResponseLog(data);
        if (response.ok) {
          showNotification('success', `SUCCESS (${response.status})`, 'New API route created successfully.');
        } else {
          showNotification('error', `ERROR (${response.status})`, data.message || 'Failed to create API.');
        }
      }
    } catch (err) {
      setResponseLog({ error: err.message });
      showNotification('error', 'ERROR', err.message);
    } finally {
      setLoading(false);
    }
  };

  // 3. Handle Download Spec Submission (POST)
  const handleDownloadSubmit = async (e) => {
    e.preventDefault();
    setLoading(true);
    setResponseLog(null);
    clearNotification();

    try {
      if (useMock) {
        await new Promise((resolve) => setTimeout(resolve, 800));
        const mockYaml = `openapi: 3.0.1\ninfo:\n  title: ${downloadForm.apiId || 'Sample_API'}\n  version: 1.0.0\npaths:\n  /users:\n    get:\n      summary: Fetch users list\n      responses:\n        '200':\n          description: Successful execution`;
        setResponseLog({ status: 200, format: downloadForm.format, content: mockYaml });
        showNotification('success', 'SUCCESS (200 OK)', 'Spec downloaded successfully (Mock response preview below).');
      } else {
        const response = await fetch(`${baseUrl}/v1/apis/spec/download`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(downloadForm)
        });
        const data = await response.text();
        setResponseLog({ responseText: data });
        if (response.ok) {
          showNotification('success', `SUCCESS (${response.status})`, 'Spec generated successfully.');
        } else {
          showNotification('error', `ERROR (${response.status})`, 'Failed to download specification.');
        }
      }
    } catch (err) {
      setResponseLog({ error: err.message });
      showNotification('error', 'ERROR', err.message);
    } finally {
      setLoading(false);
    }
  };

  // 4. Handle Health Check (GET)
  const handleHealthCheck = async () => {
    setLoading(true);
    setResponseLog(null);
    clearNotification();

    try {
      if (useMock) {
        await new Promise((resolve) => setTimeout(resolve, 600));
        const resData = {
          status: 'UP',
          components: {
            db: { status: 'UP', details: { database: 'PostgreSQL', validationQuery: 'isValid()' } },
            diskSpace: { status: 'UP', details: { total: 499963174912, free: 245100000000 } },
            ping: { status: 'UP' }
          },
          springBootVersion: '3.2.4',
          checkedAt: new Date().toISOString()
        };
        setResponseLog(resData);
        showNotification('success', 'SUCCESS (200 OK)', 'Spring Boot service is healthy and UP.');
      } else {
        const response = await fetch(`${baseUrl}/actuator/health`, {
          method: 'GET',
          headers: { 'Accept': 'application/json' }
        });
        const data = await response.json();
        setResponseLog(data);
        if (response.ok) {
          showNotification('success', `SUCCESS (${response.status})`, `Backend Status: ${data.status || 'UP'}`);
        } else {
          showNotification('error', `ERROR (${response.status})`, 'Backend Health check returned error status.');
        }
      }
    } catch (err) {
      setResponseLog({ error: err.message });
      showNotification('error', 'ERROR', 'Unable to reach Spring Boot Actuator endpoint. Ensure server is running.');
    } finally {
      setLoading(false);
    }
  };

  const currentApi = apiInfoData[activeTab];

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 font-sans flex flex-col">
      
      {/* HEADER SECTION */}
      <header className="bg-slate-900 border-b border-slate-800 sticky top-0 z-50 shadow-lg">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-3.5 flex flex-col md:flex-row items-center justify-between gap-4">
          
          {/* Logo and Title */}
          <div className="flex items-center space-x-3">
            <div className="h-10 w-10 rounded-xl bg-gradient-to-tr from-indigo-600 via-blue-500 to-cyan-400 flex items-center justify-center shadow-md shadow-indigo-500/20 ring-1 ring-white/20">
              <Layers className="h-6 w-6 text-white" />
            </div>
            <div>
              <div className="flex items-center space-x-2">
                <h1 className="text-xl font-bold bg-clip-text text-transparent bg-gradient-to-r from-white via-slate-200 to-slate-400">
                  APINexus Portal
                </h1>
                <span className="text-xs px-2 py-0.5 rounded-full bg-indigo-500/10 text-indigo-400 border border-indigo-500/20 font-medium">
                  Spring Boot Gateway
                </span>
              </div>
              <p className="text-xs text-slate-400">Enterprise Specification & Control Dashboard</p>
            </div>
          </div>

          {/* Backend Configuration Controls */}
          <div className="flex flex-wrap items-center gap-3 text-xs bg-slate-950 p-2 rounded-lg border border-slate-800">
            <div className="flex items-center space-x-2">
              <Server className="w-3.5 h-3.5 text-slate-400" />
              <span className="text-slate-400 font-medium">Base URL:</span>
              <input
                type="text"
                value={baseUrl}
                onChange={(e) => setBaseUrl(e.target.value)}
                disabled={useMock}
                className="bg-slate-900 text-slate-200 px-2 py-1 rounded border border-slate-700 text-xs w-48 focus:outline-none focus:border-indigo-500 disabled:opacity-50"
              />
            </div>

            <div className="h-4 w-px bg-slate-800 hidden sm:block"></div>

            {/* Mock Toggle */}
            <div 
              onClick={() => setUseMock(!useMock)}
              className="flex items-center space-x-2 cursor-pointer select-none"
            >
              <span className="text-slate-300 text-xs font-medium">Mock Backend:</span>
              <button
                type="button"
                className={`relative inline-flex h-5 w-9 flex-shrink-0 cursor-pointer rounded-full border-2 border-transparent transition-colors duration-200 ease-in-out focus:outline-none ${
                  useMock ? 'bg-indigo-600' : 'bg-slate-700'
                }`}
              >
                <span
                  className={`pointer-events-none inline-block h-4 w-4 transform rounded-full bg-white shadow ring-0 transition duration-200 ease-in-out ${
                    useMock ? 'translate-x-4' : 'translate-x-0'
                  }`}
                />
              </button>
            </div>
          </div>
        </div>
      </header>

      {/* Toast Notification Area */}
      {notification && (
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 pt-4 w-full">
          <div
            className={`p-4 rounded-xl border flex items-start justify-between shadow-lg transition-all ${
              notification.type === 'success'
                ? 'bg-emerald-950/80 border-emerald-500/50 text-emerald-200'
                : 'bg-rose-950/80 border-rose-500/50 text-rose-200'
            }`}
          >
            <div className="flex items-start space-x-3">
              {notification.type === 'success' ? (
                <CheckCircle2 className="w-5 h-5 text-emerald-400 mt-0.5 flex-shrink-0" />
              ) : (
                <XCircle className="w-5 h-5 text-rose-400 mt-0.5 flex-shrink-0" />
              )}
              <div>
                <h4 className="font-semibold text-sm">{notification.message}</h4>
                {notification.details && <p className="text-xs opacity-90 mt-1">{notification.details}</p>}
              </div>
            </div>
            <button
              onClick={clearNotification}
              className="text-slate-400 hover:text-white p-1 transition-colors"
            >
              ✕
            </button>
          </div>
        </div>
      )}

      {/* MAIN 3-COLUMN CONTENT LAYOUT */}
      <main className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6 flex-1 w-full grid grid-cols-1 lg:grid-cols-12 gap-6">
        
        {/* COLUMN 1: LEFT SIDE - Action Ref Links (3 cols on lg) */}
        <section className="lg:col-span-3 flex flex-col space-y-3">
          <div className="bg-slate-900 border border-slate-800 rounded-xl p-4 shadow-sm">
            <h2 className="text-xs font-semibold text-slate-400 uppercase tracking-wider mb-3 px-1">
              API Actions / Endpoints
            </h2>
            
            <nav className="space-y-2">
              {/* a) Create New API */}
              <button
                onClick={() => setActiveTab('createApi')}
                className={`w-full text-left p-3 rounded-lg flex items-center space-x-3 transition-all ${
                  activeTab === 'createApi'
                    ? 'bg-indigo-600/10 border border-indigo-500/40 text-indigo-300 font-medium shadow-inner'
                    : 'bg-slate-800/40 border border-transparent text-slate-300 hover:bg-slate-800 hover:text-white'
                }`}
              >
                <PlusCircle className={`w-4 h-4 ${activeTab === 'createApi' ? 'text-indigo-400' : 'text-slate-400'}`} />
                <div className="flex-1 truncate">
                  <span className="text-sm block truncate">Create New API</span>
                  <span className="text-[10px] text-slate-500 font-mono">POST /v1/apis/create</span>
                </div>
              </button>

              {/* b) Update Spec API */}
              <button
                onClick={() => setActiveTab('updateSpec')}
                className={`w-full text-left p-3 rounded-lg flex items-center space-x-3 transition-all ${
                  activeTab === 'updateSpec'
                    ? 'bg-indigo-600/10 border border-indigo-500/40 text-indigo-300 font-medium shadow-inner'
                    : 'bg-slate-800/40 border border-transparent text-slate-300 hover:bg-slate-800 hover:text-white'
                }`}
              >
                <UploadCloud className={`w-4 h-4 ${activeTab === 'updateSpec' ? 'text-indigo-400' : 'text-slate-400'}`} />
                <div className="flex-1 truncate">
                  <span className="text-sm block truncate">Update Spec API</span>
                  <span className="text-[10px] text-slate-500 font-mono">POST /v1/apis/spec</span>
                </div>
              </button>

              {/* c) Download Spec */}
              <button
                onClick={() => setActiveTab('downloadSpec')}
                className={`w-full text-left p-3 rounded-lg flex items-center space-x-3 transition-all ${
                  activeTab === 'downloadSpec'
                    ? 'bg-indigo-600/10 border border-indigo-500/40 text-indigo-300 font-medium shadow-inner'
                    : 'bg-slate-800/40 border border-transparent text-slate-300 hover:bg-slate-800 hover:text-white'
                }`}
              >
                <Download className={`w-4 h-4 ${activeTab === 'downloadSpec' ? 'text-indigo-400' : 'text-slate-400'}`} />
                <div className="flex-1 truncate">
                  <span className="text-sm block truncate">Download Spec</span>
                  <span className="text-[10px] text-slate-500 font-mono">POST /v1/apis/download</span>
                </div>
              </button>

              {/* d) Health Check */}
              <button
                onClick={() => setActiveTab('healthCheck')}
                className={`w-full text-left p-3 rounded-lg flex items-center space-x-3 transition-all ${
                  activeTab === 'healthCheck'
                    ? 'bg-indigo-600/10 border border-indigo-500/40 text-indigo-300 font-medium shadow-inner'
                    : 'bg-slate-800/40 border border-transparent text-slate-300 hover:bg-slate-800 hover:text-white'
                }`}
              >
                <Activity className={`w-4 h-4 ${activeTab === 'healthCheck' ? 'text-indigo-400' : 'text-slate-400'}`} />
                <div className="flex-1 truncate">
                  <span className="text-sm block truncate">Health Check</span>
                  <span className="text-[10px] text-slate-500 font-mono">GET /actuator/health</span>
                </div>
              </button>
            </nav>
          </div>

          {/* Quick Connection Info Box */}
          <div className="bg-slate-900/60 border border-slate-800 rounded-xl p-4 text-xs space-y-2">
            <div className="flex items-center justify-between text-slate-400">
              <span className="font-semibold">Backend Target:</span>
              <span className={`px-2 py-0.5 rounded text-[10px] ${useMock ? 'bg-amber-500/10 text-amber-400 border border-amber-500/20' : 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/20'}`}>
                {useMock ? 'MOCK ENGINE' : 'LIVE SPRING BOOT'}
              </span>
            </div>
            <p className="text-slate-500 leading-relaxed">
              Submissions interact directly with the specified Spring Boot backend controller methods.
            </p>
          </div>
        </section>

        {/* COLUMN 2: CENTER - Feature Information Display (4 cols on lg) */}
        <section className="lg:col-span-4 bg-slate-900 border border-slate-800 rounded-xl p-5 flex flex-col justify-between shadow-sm">
          <div>
            <div className="flex items-center justify-between pb-3 border-b border-slate-800">
              <span className="text-xs font-semibold text-slate-400 uppercase tracking-wider flex items-center gap-1.5">
                <Info className="w-3.5 h-3.5 text-indigo-400" />
                API Feature Specs
              </span>
              <span className={`px-2 py-0.5 rounded font-mono font-bold text-xs ${
                currentApi.method === 'POST' 
                  ? 'bg-blue-500/10 text-blue-400 border border-blue-500/30' 
                  : 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/30'
              }`}>
                {currentApi.method}
              </span>
            </div>

            <h3 className="text-lg font-bold text-white mt-4">{currentApi.title}</h3>
            
            <div className="mt-2 text-xs font-mono bg-slate-950 p-2.5 rounded border border-slate-800/80 text-indigo-300 break-all">
              {baseUrl}{currentApi.endpoint}
            </div>

            <p className="text-sm text-slate-300 leading-relaxed mt-4">
              {currentApi.description}
            </p>

            <div className="mt-6 space-y-4">
              <div>
                <h4 className="text-xs font-semibold text-slate-400 uppercase tracking-wide mb-2">
                  Key Capability Highlights
                </h4>
                <ul className="space-y-2">
                  {currentApi.features.map((feat, idx) => (
                    <li key={idx} className="flex items-start text-xs text-slate-300 space-x-2">
                      <span className="h-1.5 w-1.5 rounded-full bg-indigo-400 mt-1.5 flex-shrink-0" />
                      <span>{feat}</span>
                    </li>
                  ))}
                </ul>
              </div>

              <div>
                <h4 className="text-xs font-semibold text-slate-400 uppercase tracking-wide mb-2">
                  Required Headers
                </h4>
                <div className="space-y-1">
                  {currentApi.headersRequired.map((hdr, idx) => (
                    <div key={idx} className="text-[11px] font-mono bg-slate-950/60 px-2.5 py-1 rounded text-slate-400 border border-slate-800/60">
                      {hdr}
                    </div>
                  ))}
                </div>
              </div>
            </div>
          </div>

          <div className="pt-4 mt-6 border-t border-slate-800 text-[11px] text-slate-500 flex items-center justify-between">
            <span>Payload Type: JSON / Multipart</span>
            <span>Status 200 / 4xx / 5xx</span>
          </div>
        </section>

        {/* COLUMN 3: RIGHT SIDE - Dynamic Form Area (5 cols on lg) */}
        <section className="lg:col-span-5 flex flex-col space-y-6">
          <div className="bg-slate-900 border border-slate-800 rounded-xl p-5 shadow-sm">
            <div className="flex items-center justify-between pb-3 border-b border-slate-800 mb-4">
              <h3 className="text-sm font-semibold text-white flex items-center gap-2">
                <FileCode className="w-4 h-4 text-indigo-400" />
                {currentApi.title} Form
              </h3>
              <span className="text-xs text-slate-400">Spring Boot Entrypoint</span>
            </div>

            {/* FORM 1: UPDATE SPEC API (Specific requested requirement) */}
            {activeTab === 'updateSpec' && (
              <form onSubmit={handleUpdateSpecSubmit} className="space-y-4">
                
                {/* xHeaders Dropdown */}
                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">
                    xHeaders <span className="text-rose-400">*</span>
                  </label>
                  <select
                    value={updateSpecForm.xHeaders}
                    onChange={(e) => setUpdateSpecForm({ ...updateSpecForm, xHeaders: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500 focus:ring-1 focus:ring-indigo-500"
                  >
                    <option value="Yes">Yes</option>
                    <option value="No">No</option>
                  </select>
                  <p className="text-[10px] text-slate-500 mt-1">Enable custom header propagation in Spring Boot request context.</p>
                </div>

                {/* CommitMessage Text */}
                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">
                    CommitMessage
                  </label>
                  <input
                    type="text"
                    placeholder="e.g. Updated OpenAPI 3.0 spec for v2 release"
                    value={updateSpecForm.commitMessage}
                    onChange={(e) => setUpdateSpecForm({ ...updateSpecForm, commitMessage: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500 focus:ring-1 focus:ring-indigo-500 placeholder-slate-600"
                  />
                </div>

                {/* SPNValue Dropdown */}
                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">
                    SPNValue <span className="text-rose-400">*</span>
                  </label>
                  <select
                    value={updateSpecForm.spnValue}
                    onChange={(e) => setUpdateSpecForm({ ...updateSpecForm, spnValue: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500 focus:ring-1 focus:ring-indigo-500"
                  >
                    <option value="SPN">SPN (Service Principal Name)</option>
                    <option value="UAMI">UAMI (User Assigned Managed Identity)</option>
                  </select>
                </div>

                {/* yamlFile Browse Input */}
                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">
                    yamlFile Upload (.yaml / .yml) <span className="text-rose-400">*</span>
                  </label>
                  <div className="relative border-2 border-dashed border-slate-700 hover:border-indigo-500 rounded-lg p-4 text-center bg-slate-950/50 transition-colors">
                    <input
                      type="file"
                      accept=".yaml,.yml"
                      onChange={(e) => setUpdateSpecForm({ ...updateSpecForm, yamlFile: e.target.files[0] })}
                      className="absolute inset-0 w-full h-full opacity-0 cursor-pointer"
                    />
                    <UploadCloud className="w-6 h-6 text-slate-400 mx-auto mb-1" />
                    <span className="text-xs text-slate-300 block font-medium">
                      {updateSpecForm.yamlFile ? updateSpecForm.yamlFile.name : 'Click or drag YAML file to browse'}
                    </span>
                    <span className="text-[10px] text-slate-500 block mt-0.5">
                      {updateSpecForm.yamlFile ? `${(updateSpecForm.yamlFile.size / 1024).toFixed(1)} KB` : 'Supports .yaml or .yml files'}
                    </span>
                  </div>
                </div>

                {/* Submit Button */}
                <button
                  type="submit"
                  disabled={loading}
                  className="w-full bg-gradient-to-r from-indigo-600 to-blue-600 hover:from-indigo-500 hover:to-blue-500 text-white font-medium py-2.5 px-4 rounded-lg text-sm flex items-center justify-center space-x-2 transition-all shadow-md shadow-indigo-600/20 disabled:opacity-50"
                >
                  {loading ? (
                    <RefreshCw className="w-4 h-4 animate-spin" />
                  ) : (
                    <>
                      <Send className="w-4 h-4" />
                      <span>Submit Update Spec API (POST)</span>
                    </>
                  )}
                </button>
              </form>
            )}

            {/* FORM 2: CREATE NEW API */}
            {activeTab === 'createApi' && (
              <form onSubmit={handleCreateApiSubmit} className="space-y-4">
                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">API Name</label>
                  <input
                    type="text"
                    required
                    placeholder="e.g. Payment Gateway API"
                    value={createForm.apiName}
                    onChange={(e) => setCreateForm({ ...createForm, apiName: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500"
                  />
                </div>

                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">Description</label>
                  <textarea
                    rows="2"
                    placeholder="Brief overview of API purpose"
                    value={createForm.description}
                    onChange={(e) => setCreateForm({ ...createForm, description: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500"
                  ></textarea>
                </div>

                <div className="grid grid-cols-2 gap-3">
                  <div>
                    <label className="block text-xs font-medium text-slate-300 mb-1">Version</label>
                    <input
                      type="text"
                      value={createForm.version}
                      onChange={(e) => setCreateForm({ ...createForm, version: e.target.value })}
                      className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500"
                    />
                  </div>
                  <div>
                    <label className="block text-xs font-medium text-slate-300 mb-1">Auth Type</label>
                    <select
                      value={createForm.authType}
                      onChange={(e) => setCreateForm({ ...createForm, authType: e.target.value })}
                      className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500"
                    >
                      <option value="Bearer Token">Bearer Token</option>
                      <option value="OAuth2">OAuth2</option>
                      <option value="API Key">API Key</option>
                      <option value="None">None</option>
                    </select>
                  </div>
                </div>

                <button
                  type="submit"
                  disabled={loading}
                  className="w-full bg-gradient-to-r from-indigo-600 to-blue-600 hover:from-indigo-500 hover:to-blue-500 text-white font-medium py-2.5 px-4 rounded-lg text-sm flex items-center justify-center space-x-2 transition-all shadow-md shadow-indigo-600/20 disabled:opacity-50"
                >
                  {loading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <> <Send className="w-4 h-4" /> <span>Create API (POST)</span> </>}
                </button>
              </form>
            )}

            {/* FORM 3: DOWNLOAD SPEC */}
            {activeTab === 'downloadSpec' && (
              <form onSubmit={handleDownloadSubmit} className="space-y-4">
                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">Target API ID / Slug</label>
                  <input
                    type="text"
                    required
                    placeholder="e.g. user-service-v1"
                    value={downloadForm.apiId}
                    onChange={(e) => setDownloadForm({ ...downloadForm, apiId: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500"
                  />
                </div>

                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">Output Format</label>
                  <select
                    value={downloadForm.format}
                    onChange={(e) => setDownloadForm({ ...downloadForm, format: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-indigo-500"
                  >
                    <option value="yaml">YAML (.yaml)</option>
                    <option value="json">JSON (.json)</option>
                  </select>
                </div>

                <button
                  type="submit"
                  disabled={loading}
                  className="w-full bg-gradient-to-r from-indigo-600 to-blue-600 hover:from-indigo-500 hover:to-blue-500 text-white font-medium py-2.5 px-4 rounded-lg text-sm flex items-center justify-center space-x-2 transition-all shadow-md shadow-indigo-600/20 disabled:opacity-50"
                >
                  {loading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <> <Download className="w-4 h-4" /> <span>Download Spec (POST)</span> </>}
                </button>
              </form>
            )}

            {/* FORM 4: HEALTH CHECK */}
            {activeTab === 'healthCheck' && (
              <div className="space-y-4">
                <p className="text-xs text-slate-300">
                  Click the button below to send an HTTP GET request to Spring Boot's actuator health check endpoint.
                </p>

                <div className="bg-slate-950 p-3 rounded-lg border border-slate-800 text-xs font-mono text-slate-400 space-y-1">
                  <div>GET {baseUrl}/actuator/health</div>
                  <div className="text-slate-600">Expected Response: &#123; "status": "UP" &#125;</div>
                </div>

                <button
                  onClick={handleHealthCheck}
                  disabled={loading}
                  className="w-full bg-gradient-to-r from-emerald-600 to-teal-600 hover:from-emerald-500 hover:to-teal-500 text-white font-medium py-2.5 px-4 rounded-lg text-sm flex items-center justify-center space-x-2 transition-all shadow-md shadow-emerald-600/20 disabled:opacity-50"
                >
                  {loading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <> <Activity className="w-4 h-4" /> <span>Execute Health Check (GET)</span> </>}
                </button>
              </div>
            )}
          </div>

          {/* Response Payload Viewer Terminal */}
          <div className="bg-slate-900 border border-slate-800 rounded-xl p-4 shadow-sm flex-1 flex flex-col">
            <div className="flex items-center justify-between pb-2 mb-2 border-b border-slate-800">
              <span className="text-xs font-semibold text-slate-400 flex items-center gap-1.5 font-mono">
                <Terminal className="w-3.5 h-3.5 text-indigo-400" />
                Backend Response Payload
              </span>
              {responseLog && (
                <button
                  onClick={() => setResponseLog(null)}
                  className="text-[10px] text-slate-500 hover:text-slate-300 transition-colors"
                >
                  Clear console
                </button>
              )}
            </div>

            <div className="bg-slate-950 rounded-lg p-3 border border-slate-800 font-mono text-xs text-slate-300 min-h-[140px] max-h-[220px] overflow-auto flex-1">
              {loading ? (
                <div className="flex items-center justify-center h-full text-slate-500 space-x-2">
                  <RefreshCw className="w-4 h-4 animate-spin text-indigo-400" />
                  <span>Executing Spring Boot HTTP call...</span>
                </div>
              ) : responseLog ? (
                <pre className="whitespace-pre-wrap break-all text-emerald-400">
                  {typeof responseLog === 'object' ? JSON.stringify(responseLog, null, 2) : responseLog}
                </pre>
              ) : (
                <div className="text-slate-600 flex flex-col items-center justify-center h-full text-center py-4">
                  <span>No response yet.</span>
                  <span className="text-[10px]">Submit an API action form to display the response payload here.</span>
                </div>
              )}
            </div>
          </div>
        </section>

      </main>

      {/* FOOTER */}
      <footer className="bg-slate-900 border-t border-slate-800 py-3 text-center text-xs text-slate-500">
        APINexus Portal &bull; Spring Boot REST API Client &bull; Standard JSON & Multipart YAML Support
      </footer>
    </div>
  );
}
